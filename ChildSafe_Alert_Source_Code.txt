document.addEventListener('DOMContentLoaded', () => {
  // Live alert count
  const stats = CSA.getStats();
  const el = document.getElementById('liveAlertCount');
  if(el) el.textContent = `${stats.active} Active Alerts Right Now`;

  // Animate stat counters
  const counters = [
    { id: 'statActive',     val: stats.active },
    { id: 'statReunited',   val: 247 },
    { id: 'statVolunteers', val: stats.volunteers },
    { id: 'statEvents',     val: stats.events },
  ];
  counters.forEach(c => {
    const el = document.getElementById(c.id);
    if(el) { el.dataset.target = c.val; el.dataset.counter = '1'; }
  });
  initCounters();

  // Recent alerts
  const cases = CSA.getCases().filter(c => ['active','searching','investigating'].includes(c.status)).slice(0, 3);
  const grid = document.getElementById('recentAlerts');
  if(grid) {
    if(cases.length === 0) {
      grid.innerHTML = '<div class="empty-state"><div class="empty-state-icon"><i class="fa-solid fa-check-circle"></i></div><h3>No Active Alerts</h3><p>All children have been reunited.</p></div>';
    } else {
      grid.innerHTML = cases.map(c => `
        <div class="case-card${c.priority?' priority':''}">
          <div class="case-card-photo">
            <div class="photo-placeholder" style="font-size:4rem;background:${getAvatarColor(c.child_name)}22">
              <span style="font-size:3.5rem">${c.photo_emoji||'👤'}</span>
            </div>
            <div class="case-card-status"><span class="badge ${CSA.statusClass(c.status)}"><span class="status-dot ${c.status}"></span>${CSA.statusLabel(c.status)}</span></div>
          </div>
          <div class="case-card-body">
            <div class="case-id">${c.case_id}</div>
            <div class="case-name">${c.child_name}</div>
            <div class="case-meta">
              <span><i class="fa-solid fa-cake-candles"></i> Age ${c.age}</span>
              <span><i class="fa-solid fa-venus-mars"></i> ${c.gender}</span>
            </div>
            <div class="case-location"><i class="fa-solid fa-location-dot" style="color:var(--accent)"></i> ${c.last_seen_location}</div>
            <div class="case-time" style="margin-top:.25rem"><i class="fa-regular fa-clock"></i> Reported ${CSA.timeAgo(c.reported_at)}</div>
          </div>
          <div class="case-card-footer">
            <a href="case.html?id=${c.case_id}" class="btn btn-primary btn-sm" style="width:100%"><i class="fa-solid fa-eye"></i> View Details</a>
          </div>
        </div>`).join('');
    }
  }
});
